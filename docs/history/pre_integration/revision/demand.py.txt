"""Explicit normalized profiles and fully materialized exogenous demand."""
import csv
from pathlib import Path
import numpy as np
from revision.core import ROOT, digest, file_hash, write_json

ORDER = ['N_to_S', 'S_to_N', 'W_to_E', 'E_to_W', 'NW_to_SE', 'SE_to_NW',
         'SW_to_NE', 'NE_to_SW', 'Periphery_to_Center', 'Center_to_Periphery', 'Uniform']


def profiles(network):
    names = sorted(ORDER) if network == 'grid' else ORDER
    directory = ROOT / ('data_traffic' if network == 'grid' else 'real_net_subnet/demand_groups')
    total = 3000. if network == 'grid' else 2383.3333
    result = []
    for name in names:
        path = directory / (('demand_' if network == 'grid' else '') + name + '.csv')
        with path.open() as stream:
            raw = [(r['origin_edge'], r['dest_edge'], float(r['veh_per_hour']))
                   for r in csv.DictReader(stream)]
        rates = np.array([r[2] for r in raw])
        if not np.all(np.isfinite(rates)) or np.any(rates < 0) or rates.sum() <= 0:
            raise ValueError('Invalid demand profile: ' + str(path))
        scale = total / rates.sum()
        result.append({'name': name, 'source': str(path), 'source_hash': file_hash(path),
                       'original_total': float(rates.sum()), 'scale': float(scale),
                       'rows': [(o, d, float(r * scale)) for o, d, r in raw]})
    return result


def mixture(groups, weights):
    weights = np.asarray(weights, dtype=float)
    if len(weights) != 11 or not np.all(np.isfinite(weights)) or np.any(weights < 0) or not np.isclose(weights.sum(), 1):
        raise ValueError('Expected eleven normalized mixture weights')
    rows = {}
    for group, weight in zip(groups, weights):
        for origin, destination, rate in group['rows']:
            key = (origin, destination)
            rows[key] = rows.get(key, 0.) + weight * rate
    return [(o, d, r) for (o, d), r in sorted(rows.items()) if r > 0]


def materialize(rows, start, duration, rng, route_resolver, prefix):
    vehicles = []
    for origin, destination, rate in rows:
        # Validate routes even when this particular Poisson realization has zero arrivals.
        edges = list(route_resolver(origin, destination))
        if not edges or edges[0] != origin or edges[-1] != destination:
            raise ValueError('Unreachable OD: {} -> {}'.format(origin, destination))
        count = rng.poisson(rate * duration / 3600.)
        times = rng.uniform(start, start + duration, count) + rng.normal(0, 2., count)
        times = np.clip(times, start + .01, start + duration - .01)
        for departure in sorted(times):
            speed = float(rng.normal(1., .1))
            while speed <= 0:
                speed = float(rng.normal(1., .1))
            vehicles.append({'id': '{}_{}'.format(prefix, len(vehicles)),
                             'depart': round(float(departure), 2), 'origin': origin,
                             'destination': destination, 'edges': edges, 'speed_factor': speed})
    return sorted(vehicles, key=lambda v: (v['depart'], v['id']))


def save_artifact(path, vehicles, metadata):
    artifact = dict(metadata, vehicles=vehicles)
    artifact['hash'] = digest(artifact)
    write_json(path, artifact)
    return artifact


def check_artifact(artifact):
    content = dict(artifact)
    expected = content.pop('hash')
    if digest(content) != expected:
        raise ValueError('Demand artifact hash mismatch')
    ids = [v['id'] for v in content['vehicles']]
    if len(set(ids)) != len(ids):
        raise ValueError('Duplicate vehicle identifiers')
    return content
