"""Auditable rollout statistics and geometry-based controlled-approach maps."""
import csv
import json
import math
import xml.etree.ElementTree as ET
from collections import defaultdict
from pathlib import Path
import numpy as np
from experiments.core import ROOT, digest, write_json
from experiments.protocol import settings


def statistics(values):
    a=np.asarray(values,dtype=float)
    return {'n':len(a),'mean':float(a.mean()) if len(a) else None,
            'sd':float(a.std(ddof=1)) if len(a)>1 else None}


def seed_interval(values):
    out=statistics(values)
    out['ci95']=2.776*out['sd']/math.sqrt(5) if out['n']==5 else None
    out['complete']=out['n']==5
    return out


def csv_write(path, rows):
    fields=sorted(set(k for r in rows for k in r))
    with path.open('x',newline='') as f:
        w=csv.DictWriter(f,fieldnames=fields); w.writeheader()
        for r in rows: w.writerow({k:json.dumps(v) if isinstance(v,(dict,list)) else v for k,v in r.items()})


def collect(roots):
    rows=[]; rejected=[]; seen={};traffic={};policies={}
    for root in roots:
        for path in sorted(root.rglob('rollout_summary.json')):
            try:
                r=json.loads(path.read_text()); manifest=json.loads((path.parent/'manifest.json').read_text())
                m=manifest; body=dict(m); claimed=body.pop('manifest_hash')
                if digest(body)!=claimed: raise ValueError('Manifest hash mismatch')
                result=json.loads((path.parent/'result.json').read_text())
                if result.get('status')!='complete' or result.get('manifest_hash')!=claimed: raise ValueError('Incomplete result')
                if r.get('status')!='complete' or r.get('sample_count')!=3600: raise ValueError('Incomplete horizon')
                artifact=json.loads(Path(m['artifact']).read_text())
                from experiments.demand import check_artifact
                check_artifact(artifact)
                if artifact['hash']!=r['demand_hash']: raise ValueError('Demand hash mismatch')
                scenario=artifact.get('scenario',{'id':'pilot_uniform','split':'pilot','family':'pilot'})
                row=dict(r,network=m['network'],controller=m['controller'],method=m['method'],seed=m['seed'],
                         pilot=m.get('pilot',False),scenario=scenario['id'],split=scenario['split'],family=scenario['family'],
                         arrival_seed=artifact['arrival_seed'],policy_seed=m['policy_seed'],wall_seconds=result['wall_seconds'],
                         record=str(path),manifest_hash=claimed)
                key=tuple(row[k] for k in ('network','controller','method','seed','scenario','split','arrival_seed','pilot'))
                if key in seen: raise ValueError('Duplicate comparison observation; select one explicit attempt')
                tk=(row['network'],row['scenario'],row['split'],row['arrival_seed'],row['pilot'])
                tv=(row['demand_hash'],row['effective_sumo_seed'])
                if tk in traffic and traffic[tk]!=tv:raise ValueError('Unpaired demand or SUMO seed')
                pk=tk+(row['controller'],row['seed'])
                if pk in policies and policies[pk]!=row['policy_seed']:raise ValueError('Unpaired policy seed')
                traffic[tk]=tv;policies[pk]=row['policy_seed']
                seen[key]=str(path);rows.append(row)
            except Exception as exc: rejected.append({'path':str(path),'reason':str(exc)})
    return rows,rejected


def summarize(rows):
    buckets=defaultdict(list)
    for r in rows:
        buckets[tuple(r[k] for k in ('network','controller','method','seed','split','scenario','family','pilot'))].append(r)
    scenario_rows=[]
    for key,rr in sorted(buckets.items()):
        row=dict(zip(('network','controller','method','seed','split','scenario','family','pilot'),key))
        row.update(statistics([r['mean_queue'] for r in rr]))
        row['complete']=len(rr)==10 and {r['arrival_seed'] for r in rr}==set(settings()['arrival_seeds'])
        scenario_rows.append(row)
    seeds=[]
    for network in settings()['networks']:
        for controller in settings()['controllers']:
            for method in settings()['methods']:
                for seed in settings()['training_seeds']:
                    for split,expected in [('test',12),('seen',11)]:
                        rr=[r for r in scenario_rows if (r['network'],r['controller'],r['method'],r['seed'],r['split'],r['pilot'])==(network,controller,method,seed,split,False)]
                        complete=len(rr)==expected and all(r['complete'] for r in rr)
                        seeds.append(dict(network=network,controller=controller,method=method,seed=seed,split=split,
                                          complete=complete,scenarios=len(rr),mean=float(np.mean([r['mean'] for r in rr])) if complete else None))
    aggregates=[]; paired=[]
    for network in settings()['networks']:
        for controller in settings()['controllers']:
            for split in ('test','seen'):
                for method in settings()['methods']:
                    rr=[r for r in seeds if r['network']==network and r['controller']==controller and r['method']==method and r['split']==split and r['complete']]
                    aggregates.append(dict(seed_interval([r['mean'] for r in rr]),network=network,controller=controller,method=method,split=split))
                    if method=='online_wce': continue
                    online={r['seed']:r['mean'] for r in seeds if (r['network'],r['controller'],r['method'],r['split'],r['complete'])==(network,controller,'online_wce',split,True)}
                    differences=[online[r['seed']]-r['mean'] for r in rr if r['seed'] in online]
                    paired.append(dict(seed_interval(differences),network=network,controller=controller,comparator=method,split=split,direction='online_wce minus comparator'))
    return scenario_rows,seeds,aggregates,paired


def lane_window(npz, start=1200, end=2400):
    # Observations timestamp the end of [t-1,t); select intervals in [start,end).
    with np.load(str(npz)) as data:
        times=data['time']; selected=(times>start)&(times<=end)
        if selected.sum()!=end-start: raise ValueError('Incomplete heatmap window')
        q=data['queue'][selected]; lanes=data['lanes'].tolist()
        return lanes,q.mean(axis=0),float(q.sum(axis=1).mean())


def heatmaps(rows, output):
    import matplotlib
    matplotlib.use('Agg')
    import matplotlib.pyplot as plt
    from matplotlib.collections import LineCollection
    selected=[r for r in rows if r['scenario'] in ('peak_1.25','peak_1.5')]
    groups=defaultdict(list)
    for r in selected: groups[(r['network'],r['controller'],r['scenario'],r['pilot'])].append(r)
    maps=[]
    for key,rr in groups.items():
        network,controller,scenario,pilot=key
        by_method={m:[r for r in rr if r['method']==m] for m in settings()['methods']}
        ids=[{(r['seed'],r['arrival_seed']) for r in by_method[m]} for m in settings()['methods']]
        paired=set.intersection(*ids) if ids else set()
        if not paired: continue
        values={}; lanes=None
        for method,records in by_method.items():
            arrays=[]
            for r in records:
                if (r['seed'],r['arrival_seed']) not in paired: continue
                ll,v,total=lane_window(Path(r['record']).parent/'rollout.npz')
                if lanes is not None and ll!=lanes: raise ValueError('Heatmap lane order mismatch')
                if not np.isclose(v.sum(),total): raise ValueError('Heatmap queue reconciliation failed')
                lanes=ll;arrays.append(v)
            values[method]=np.mean(arrays,axis=0)
        net=ROOT/('large_grid/data/exp.net.xml' if network=='grid' else 'real_net_subnet/data/in/most.net.xml')
        geometry={}
        for edge in ET.parse(str(net)).getroot().findall('edge'):
            if edge.get('function')=='internal': continue
            for lane in edge.findall('lane'):
                geometry[lane.get('id')]=[[float(x) for x in point.split(',')[:2]] for point in lane.get('shape').split()]
        vmax=max(float(v.max()) for v in values.values()) or 1.
        differences={m:values['online_wce']-values[m] for m in settings()['methods'] if m!='online_wce'}
        dmax=max(float(np.abs(v).max()) for v in differences.values()) or 1.
        prefix='{}_{}_{}_{}'.format(network,controller,scenario,'pilot' if pilot else 'publication')
        for name,v,diff in [(m,v,False) for m,v in values.items()]+[('online_minus_'+m,v,True) for m,v in differences.items()]:
            fig,ax=plt.subplots(figsize=(8,6));ax.add_collection(LineCollection(list(geometry.values()),colors='#c5cbd1',linewidths=.7))
            collection=LineCollection([geometry[l] for l in lanes],cmap='RdBu_r' if diff else 'viridis',linewidths=2)
            collection.set_array(v);collection.set_clim(-dmax if diff else 0,dmax if diff else vmax);ax.add_collection(collection)
            ax.autoscale();ax.set_aspect('equal');ax.axis('off');ax.set_title(name+' | '+scenario+' | '+str(len(paired))+' paired observations')
            fig.colorbar(collection,ax=ax,label='Mean stopped vehicles per monitored lane');fig.tight_layout()
            for ext in ('png','svg'): fig.savefig(str(output/(prefix+'_'+name+'.'+ext)),dpi=160)
            plt.close(fig)
        maps.append({'network':network,'controller':controller,'scenario':scenario,'pilot':pilot,'pairs':len(paired),
                     'lanes':lanes,'geometry':{l:geometry[l] for l in lanes},'values':{m:v.tolist() for m,v in values.items()},
                     'absolute_max':vmax,'difference_max':dmax,'window':[1200,2400]})
    return maps


def report(roots, output):
    output.mkdir(parents=True,exist_ok=False)
    rows,rejected=collect(roots)
    # Ambiguous duplicated observations must not silently enter the report.
    if any('Duplicate comparison' in r['reason'] or 'Unpaired' in r['reason'] for r in rejected):
        write_json(output/'rejected.json',rejected)
        raise ValueError('Duplicate observations: choose explicit attempt roots; see rejected.json')
    scenarios,seeds,aggregates,paired=summarize(rows)
    for name,rr in [('rollouts',rows),('scenarios',scenarios),('seeds',seeds),('comparison',aggregates),('paired_differences',paired),('rejected',rejected)]:
        csv_write(output/(name+'.csv'),rr)
    curves=[];costs=[]
    for root in roots:
        for p in sorted(root.rglob('progress.jsonl')):
            manifest=json.loads((p.parent/'manifest.json').read_text())
            entries=[json.loads(line) for line in p.read_text().splitlines() if line.strip()]
            curves.append({'network':manifest['network'],'controller':manifest['controller'],'method':manifest['method'],
                           'pilot':manifest.get('pilot',False),'stage':manifest['stage'],'points':entries})
        for p in sorted(root.rglob('result.json')):
            mpath=p.parent/'manifest.json'
            if not mpath.exists(): continue
            m=json.loads(mpath.read_text());r=json.loads(p.read_text())
            if m.get('stage') not in ('parent','wce','continue'): continue
            costs.append({'network':m['network'],'controller':m['controller'],'method':m['method'],'stage':m['stage'],
                          'seed':m['seed'],'pilot':m.get('pilot',False),'status':r['status'],'wall_seconds':r.get('wall_seconds'),
                          'components':r.get('component_seconds',{}),'parents':m.get('parents',{}),'record':str(p)})
    csv_write(output/'computation_costs.csv',costs)
    pipeline=[]
    for cost in costs:
        if cost['stage']!='continue' or cost['status']!='complete':continue
        entry={k:cost[k] for k in ('network','controller','method','seed','pilot')}
        entry['continuation_seconds']=cost['wall_seconds']
        entry['parent_seconds']=None;entry['offline_wce_seconds']=0.
        try:
            parent=Path(cost['parents']['controller']['path']).parent/'result.json'
            entry['parent_seconds']=json.loads(parent.read_text())['wall_seconds']
            if cost['method'] in ('fixed_wce','online_wce'):
                offline=Path(cost['parents']['wce']['path']).parent/'result.json'
                entry['offline_wce_seconds']=json.loads(offline.read_text())['wall_seconds']
            entry['standalone_seconds']=entry['parent_seconds']+entry['offline_wce_seconds']+entry['continuation_seconds']
        except (KeyError,OSError,ValueError):entry['standalone_seconds']=None
        pipeline.append(entry)
    csv_write(output/'standalone_pipeline_costs.csv',pipeline)
    family_rows=[];worst=[]
    for network in settings()['networks']:
        for controller in settings()['controllers']:
            for method in settings()['methods']:
                candidates=[r for r in scenarios if (r['network'],r['controller'],r['method'],r['split'],r['pilot'],r['complete'])==(network,controller,method,'test',False,True)]
                for family in ('redistribution','mixture','temporal','peak'):
                    vals=[]
                    for seed in settings()['training_seeds']:
                        rr=[r for r in candidates if r['seed']==seed and r['family']==family]
                        if len(rr)==3:vals.append(float(np.mean([r['mean'] for r in rr])))
                    family_rows.append(dict(seed_interval(vals),network=network,controller=controller,method=method,family=family))
                means=[]
                for scenario in sorted({r['scenario'] for r in candidates}):
                    rr=[r for r in candidates if r['scenario']==scenario]
                    if {r['seed'] for r in rr}==set(settings()['training_seeds']):means.append((scenario,float(np.mean([r['mean'] for r in rr]))))
                if len(means)==12:
                    name,value=max(means,key=lambda x:x[1]);worst.append(dict(network=network,controller=controller,method=method,scenario=name,worst_tested_scenario_mean=value))
    csv_write(output/'demand_family_summary.csv',family_rows);csv_write(output/'worst_tested_scenario.csv',worst)
    maps=heatmaps(rows,output)
    import matplotlib
    matplotlib.use('Agg')
    import matplotlib.pyplot as plt
    for i,c in enumerate(curves):
        fig,ax=plt.subplots();ax.plot([p['learning_steps'] for p in c['points']],[p['mean_queue'] for p in c['points']])
        ax.set(xlabel='Controller-learning steps',ylabel='Mean total queue (vehicles)',title='{} / {} / {} / {}'.format(c['network'],c['controller'],c['method'],'pilot' if c['pilot'] else 'publication'))
        for ext in ('png','svg'):fig.savefig(str(output/('training_%03d.'%i+ext)),bbox_inches='tight')
        plt.close(fig)
    public_rows=[{k:v for k,v in r.items() if k not in ('record','checkpoint')} for r in rows]
    export={'version':1,'rollouts':public_rows,'scenarios':scenarios,'seeds':seeds,'comparisons':aggregates,'paired':paired,
            'curves':curves,'heatmaps':maps,'families':family_rows,'worst':worst,'costs':pipeline,
            'rejected_count':len(rejected),'publication_complete':all(r['complete'] for r in aggregates)}
    write_json(output/'dashboard.json',export)
    return {'output':str(output),'rollouts':len(rows),'rejected':len(rejected),'publication_complete':export['publication_complete']}
